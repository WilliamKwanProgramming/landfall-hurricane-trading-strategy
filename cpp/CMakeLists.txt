# pybind11 module, written to <build>/cpp/landfall_cpp*.so (tests/conftest.py looks there).
set(landfall_cpp_skip "")
if(LANDFALL_SANITIZE)
  set(landfall_cpp_skip "LANDFALL_SANITIZE=ON (the Python interpreter is not instrumented)")
else()
  find_package(Python3 COMPONENTS Interpreter Development.Module QUIET)
  if(NOT Python3_FOUND)
    set(landfall_cpp_skip "Python3 development headers not found")
  endif()
endif()
if(NOT landfall_cpp_skip AND NOT pybind11_DIR)
  execute_process(COMMAND "${Python3_EXECUTABLE}" -m pybind11 --cmakedir
    OUTPUT_VARIABLE landfall_pybind11_dir OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET
    RESULT_VARIABLE landfall_pybind11_rc)
  if(landfall_pybind11_rc EQUAL 0 AND landfall_pybind11_dir)
    set(pybind11_DIR "${landfall_pybind11_dir}" CACHE PATH "pybind11 CMake config directory")
  endif()
endif()
if(NOT landfall_cpp_skip)
  find_package(pybind11 3.0 CONFIG QUIET)
  if(NOT pybind11_FOUND)
    set(landfall_cpp_skip "pybind11 >= 3.0 not found (pip install pybind11)")
  endif()
endif()
if(landfall_cpp_skip)
  message(STATUS "landfall_cpp: skipped, ${landfall_cpp_skip}")
else()
  message(STATUS "landfall_cpp: Python ${Python3_VERSION}, pybind11 ${pybind11_VERSION}")
  pybind11_add_module(landfall_cpp MODULE src/bindings.cpp)
  target_link_libraries(landfall_cpp PRIVATE landfall_footprint)
  set_target_properties(landfall_cpp PROPERTIES LIBRARY_OUTPUT_DIRECTORY
    "${CMAKE_CURRENT_BINARY_DIR}")
endif()
